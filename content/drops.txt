# What monsters leave behind, standing in for the original's drop tables
# (Table.Tbd tables 30 and 31): see docs/content.md. Each [drops <list>] is
# used by the monsters whose enemy= line names that list (the field after
# the AI list, the level the hero should be at there). Like the original, a list has its rolls, each
# dropping something `chance` percent of the time (a little more: as in
# the original, 15 means 16 in 100), as one of its picks chosen at random.
# Monsters with no gold of their own drop gold= (chance, least, most).
# An `any` pick may give a normal, magic or unique item, as the original's
# usually may; a fifth number limits that (110: no normal items, 11: no
# uniques). How often each comes is down to the items' weight=.

[drops 1]
rolls=1
chance=15
gold=10,3,8
pick=consumable,0        # Healing Potion
pick=consumable,0
pick=consumable,10000000 # Mana Potion
pick=consumable,10000000
pick=special,1          # Mine (into the hero's mines, not the bag)
pick=any,any,1,1         # a weapon, armour or ring of item level 1-1, by weight
pick=any,any,1,1
pick=any,any,1,1
