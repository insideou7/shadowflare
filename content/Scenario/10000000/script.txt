# Remote Town's people: where every new game begins, as in the original
# (scenario 0). The talk follows the original's order in our own words:
# the commander greets the hero over several boxes and hands over the
# army's kit and a little gold, the healer tells of the monsters to the
# north-east and gives the first mission, and the smith trades.
#
# temp:2 holds which of the commander's boxes is showing, so that closing
# it shows the next one; save:4 and save:5 remember that the hero has met
# the commander and the healer.
#
# The warp gate north-east of the square, as the original's: walking up
# to it finds it (warp:0, which a new game already has) and shows its
# name over it, and a click on it opens the GATE window.

object=9361,2086,9658,2337,gate
object=8160,697,10460,3197,gate-ground

# The commander
[sentence commander]
on=talk commander
C_STOP commander
C_SETANGLETO commander 0
IF save:4 = 1 @commander-again
IF save:4 = 0 @commander-first

[sentence commander-first]
SETFLAG save:4 1
SETFLAG temp:2 1
TALK msg:1 temp:1 0 0 -1 -1

[sentence commander-again]
SETFLAG temp:2 0
TALK msg:5 temp:1 0 0 -1 -1

[sentence commander-end]
on=talk-end commander
SETFLAG temp:5 temp:2
SETFLAG temp:2 0
IF temp:5 = 0 @commander-done
IF temp:5 = 1 @commander-2
IF temp:5 = 2 @commander-3
IF temp:5 = 3 @commander-4
IF temp:5 = 4 @commander-5

[sentence commander-done]

[sentence commander-2]

[sentence commander-3]

# The healer gives the first mission the first time, and heals after.
# temp:21 says which question is showing: 1 the heal menu, 2 the
# greeting. As IF carries on after its sentence, each check comes before
# the sentence that would make it true.
[sentence healer]
on=talk healer
C_STOP healer
C_SETANGLETO healer 0
IF save:5 = 1 @healer-menu
IF save:5 = 0 @healer-first

[sentence healer-first]
SETFLAG save:5 1
SETFLAG temp:21 2
TALK msg:20 temp:20 0 0 -1 -1

[sentence healer-menu]
SETFLAG temp:21 1
TALK msg:22 temp:20 0 0 -1 -1

[sentence healer-end]
on=talk-end healer
SETFLAG temp:22 temp:21
SETFLAG temp:21 0
IF temp:22 = 0 @healer-done
IF temp:22 = 1 @healer-choice
IF temp:22 = 2 @healer-mission

[sentence healer-done]
C_MOVABLE healer

[sentence healer-mission]
SETQUESTFLAG 0 1 0
DRAWQUESTNAME 0
TALK msg:21 temp:20 0 0 -1 -1

[sentence healer-choice]
IF temp:20 = 0 @healer-heal
IF temp:20 ! 0 @healer-done

[sentence healer-heal]
# The healer casts (person chart 3) as the hero is healed.
C_ACTIONREQ healer 4 -1 -1 -1 -1
HEALLIFE
HEALMENTAL
TALK msg:23 temp:20 0 0 -1 -1
