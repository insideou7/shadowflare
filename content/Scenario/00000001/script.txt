# Goblin Meadow's villagers: see docs/content.md. The people named here
# are in scenario.txt. Each one keeps the answer to their last question in
# one TEMP flag and which question it was in another, as the original's
# scripts do, since closing any message box runs the same talk-end
# sentence.

# The elder
[sentence elder]
on=talk elder
C_STOP elder
C_SETANGLETO elder 0
TALK msg:1 temp:1 0 0 -1 -1

[sentence elder-end]
on=talk-end elder
C_MOVABLE elder

# The merchant: list 0 of shops.txt
[sentence merchant]
on=talk merchant
C_STOP merchant
C_SETANGLETO merchant 0
TALK msg:2 temp:2 0 0 -1 -1

[sentence merchant-end]
on=talk-end merchant
C_MOVABLE merchant
IF temp:2 = 0 @merchant-shop

[sentence merchant-shop]
INITTRADE 0 0
TRADE 0

# The priest heals for nothing
[sentence priest]
on=talk priest
C_STOP priest
C_SETANGLETO priest 0
SETFLAG temp:4 1
TALK msg:3 temp:3 0 0 -1 -1

[sentence priest-end]
on=talk-end priest
C_MOVABLE priest
SETFLAG temp:5 temp:4
SETFLAG temp:4 0
IF temp:5 = 1 @priest-choice

[sentence priest-choice]
IF temp:3 = 0 @priest-heal

[sentence priest-heal]
HEALLIFE
HEALMENTAL
TALK msg:4 temp:3 0 0 -1 -1

# The smith mends and identifies. temp:10 is the answer, temp:11 the
# question (1 the menu, 2 mending worn things, 3 carried things, 4
# identifying), temp:12 the price.
[sentence smith]
on=talk smith
C_STOP smith
C_SETANGLETO smith 0
SETFLAG temp:11 1
TALK msg:10 temp:10 0 0 -1 -1

[sentence smith-end]
on=talk-end smith
C_MOVABLE smith
SETFLAG temp:15 temp:11
SETFLAG temp:11 0
IF temp:15 = 1 @smith-menu
IF temp:15 = 2 @smith-pay
IF temp:15 = 3 @smith-pay
IF temp:15 = 4 @smith-pay

[sentence smith-menu]
IF temp:10 = 0 @smith-worn
IF temp:10 = 1 @smith-carried
IF temp:10 = 2 @smith-finds

[sentence smith-worn]
GETREPAIRPRICE 0 temp:20
GETREPAIRPRICE 1 temp:21
GETREPAIRPRICE 2 temp:22
GETREPAIRPRICE 3 temp:23
GETREPAIRPRICE 4 temp:24
SETFLAG temp:12 0
ADD temp:12 temp:20
ADD temp:12 temp:21
ADD temp:12 temp:22
ADD temp:12 temp:23
ADD temp:12 temp:24
SETFLAG temp:16 2
IF temp:12 = 0 @smith-nothing
IF temp:12 > 0 @smith-ask

[sentence smith-carried]
GETREPAIRPRICE -1 temp:12
SETFLAG temp:16 3
IF temp:12 = 0 @smith-nothing
IF temp:12 > 0 @smith-ask

[sentence smith-nothing]
TALK msg:12 temp:10 0 0 -1 -1

[sentence smith-ask]
SETMSGDEFINE temp:12
SETFLAG temp:11 temp:16
TALK msg:11 temp:10 0 0 -1 -1

[sentence smith-finds]
CHECKUNOPINIONITEM temp:17
IF temp:17 = 0 @smith-seen
IF temp:17 = 1 @smith-ask-finds

[sentence smith-seen]
TALK msg:15 temp:10 0 0 -1 -1

[sentence smith-ask-finds]
SETFLAG temp:12 50
SETFLAG temp:11 4
TALK msg:14 temp:10 0 0 -1 -1

# Paying: temp:15 still says what for.
[sentence smith-pay]
IF temp:10 = 0 @smith-gold

[sentence smith-gold]
GETTOTALGOLD temp:13
SETFLAG temp:18 0
IF temp:13 < temp:12 @smith-poor
IF temp:18 = 0 @smith-work

[sentence smith-poor]
SETFLAG temp:18 1
TALK msg:13 temp:10 0 0 -1 -1

[sentence smith-work]
PAYGOLD temp:12
IF temp:15 = 2 @smith-mend-worn
IF temp:15 = 3 @smith-mend-carried
IF temp:15 = 4 @smith-identify

[sentence smith-mend-worn]
REPAIR 0
REPAIR 1
REPAIR 2
REPAIR 3
REPAIR 4
TALK msg:16 temp:10 0 0 -1 -1

[sentence smith-mend-carried]
REPAIR -1
TALK msg:16 temp:10 0 0 -1 -1

[sentence smith-identify]
IDENTIFY
TALK msg:17 temp:10 0 0 -1 -1
