[sentence arrive]
on=start
SETFLAG temp:50 0
GETENTRYPOINT temp:52
IF temp:52 < 0 @arrive-none
IF temp:52 = 0 @arrive-0
IF temp:52 = 1 @arrive-1
IF temp:52 = 2 @arrive-2
IF temp:52 = 3 @arrive-3
IF temp:52 = 4 @arrive-4

[sentence arrive-none]

[sentence arrive-0]

[sentence arrive-4]

# The hero's remark floats over them, fading over its last 100 ticks;
# temp:50 counts down the ticks it has left.
[sentence remark]
on=tick
IF temp:50 ! 0 @remark-show

[sentence remark-show]
GETMYPLAYERNO temp:53
GETPLAYERSEX temp:54
SETFLAG temp:55 msg:20
IF temp:54 = 0 @remark-her
IF temp:50 < 100 @remark-fade
IF temp:50 > 99 @remark-full
MESSAGEBOX temp:53 0 -80 temp:55 224 224 224 temp:56
SUB temp:50 1

[sentence remark-her]
SETFLAG temp:55 msg:21

[sentence remark-fade]
SETFLAG temp:56 temp:50
MUL temp:56 10

[sentence remark-full]
SETFLAG temp:56 1000

# Mission 10 (search the Immortal Remains) is done once its last guards (monsters 0 to 6) are dead, as in
# the original, where the hall's sealed door opens then.
[sentence mission-check]
on=tick
IF quest:10 ! 2 @mission-look

[sentence mission-look]
CHECKENEMYLIVE 14000000 14000006 temp:51
IF temp:51 = -1 @mission-done

[sentence mission-done]
SETQUESTFLAG 10 2 1
