[sentence arrive]
on=start
SETCOMPASSES msg:1
SETFLAG temp:50 200

# The hero's remark floats over them, fading over its last 100 ticks;
# temp:50 counts down the ticks it has left.
[sentence remark]
on=tick
IF temp:50 ! 0 @remark-show

[sentence remark-show]
GETMYPLAYERNO temp:53
GETPLAYERSEX temp:54
SETFLAG temp:55 msg:20
IF temp:54 = 0 @remark-her
IF temp:50 < 100 @remark-fade
IF temp:50 > 99 @remark-full
MESSAGEBOX temp:53 0 -80 temp:55 224 224 224 temp:56
SUB temp:50 1

[sentence remark-her]
SETFLAG temp:55 msg:21

[sentence remark-fade]
SETFLAG temp:56 temp:50
MUL temp:56 10

[sentence remark-full]
SETFLAG temp:56 1000

# Mission 9 (explore past the Remains) is done on reaching this place.
[sentence mission-done]
on=start
IF quest:9 = 1 @mission-clear

[sentence mission-clear]
SETQUESTFLAG 9 2 0

# Burrows (written by tools/content/enemy_holes.py)
#
# Each burrow brings its monsters back, one at a time, while it
# stands: every 200 ticks, if one of them is dead, the burrow glows
# and it climbs back out. Breaking the burrow ends it. temp:<n> flags
# from 900 on: the tick count, the glow left, the monster
# coming back, whether the burrow is still standing, set up.

[sentence hole0-start]
on=start
IF temp:904 = 0 @hole0-setup

[sentence hole0-setup]
SETFLAG temp:904 1
SETFLAG shown:hole0 1

# Monsters 14000001 to 14000005 are this burrow's.
[sentence hole0-tick]
on=tick
CHECKENEMYLIVE 14000000 14000000 temp:903
IF temp:900 = 180 @hole0-wake
IF temp:900 = 200 @hole0-revive
IF temp:903 ! -1 @hole0-glow
ADD temp:900 1

[sentence hole0-wake]
IF temp:903 ! -1 @hole0-look

[sentence hole0-look]
CHECKENEMYDEAD 14000001 14000005 temp:902
IF temp:902 ! -1 @hole0-open

[sentence hole0-open]
SETFLAG shown:hole0 0
SETFLAG shown:hole0-open 1
SETFLAG temp:901 40
CREATEEFFECT 20007 x:hole0 y:hole0 0 -1
CREATEEFFECT 20008 x:hole0 y:hole0 0 -1
PLAYSOUND 27 0 x:hole0 y:hole0

[sentence hole0-revive]
IF temp:902 ! -1 @hole0-bring
SETFLAG temp:900 0

[sentence hole0-bring]
IF temp:903 ! -1 @hole0-rise

[sentence hole0-rise]
RESSURECT temp:902 x:hole0 y:hole0 7

[sentence hole0-glow]
IF temp:901 > -1 @hole0-fade

[sentence hole0-fade]
IF temp:901 = 0 @hole0-shut
SUB temp:901 1

[sentence hole0-shut]
SETFLAG shown:hole0 1
SETFLAG shown:hole0-open 0

[sentence hole0-broken]
on=enemy-dead 14000000
PLAYSOUND 28 0 x:hole0 y:hole0
SETFLAG shown:hole0 0
SETFLAG shown:hole0-open 0
SETFLAG shown:hole0-broken 1
